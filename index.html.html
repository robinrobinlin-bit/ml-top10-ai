Cloud Function 範例程式

真正部署時，把前端的 API URL 換成你的 Google Cloud Functions、Firebase Functions 或 Cloudflare Workers 網址。

// Frontend fetch example
async function askCloudFunction(task, question) {
  const res = await fetch('https://YOUR-CLOUD-FUNCTION-URL/recommend', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ task, question })
  });
  return await res.json();
}

// Node.js Cloud Function example
exports.recommend = async (req, res) => {
  const { task, question } = req.body;
  const map = {
    regression: 'Linear Regression / Random Forest / XGBoost',
    classification: 'Logistic Regression / Decision Tree / Random Forest / XGBoost',
    clustering: 'K-Means Clustering',
    text: 'Naive Bayes / SVM / Transformer',
    deep: 'Neural Network / CNN / Transformer'
  };
  res.json({ task, question, recommendation: map[task] || 'Decision Tree' });
};
